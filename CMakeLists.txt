cmake_minimum_required(VERSION 3.16)
project(routeline VERSION 0.16.19 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(LINE_BUILD_PLUGIN "Build the Linux Metamod server plugin" ON)
set(SDK "${CMAKE_CURRENT_SOURCE_DIR}/deps/s2sdk" CACHE PATH "Pinned CS2 S2SDK")
set(MMS "${CMAKE_CURRENT_SOURCE_DIR}/deps/metamod-source" CACHE PATH "Pinned Metamod API 18")
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/deps/zstd/build/cmake/CMakeLists.txt")
  message(FATAL_ERROR "Run bootstrap.sh first (zstd source is required).")
endif()
set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZSTD_MULTITHREAD_SUPPORT OFF CACHE BOOL "" FORCE)
set(ZSTD_LEGACY_SUPPORT OFF CACHE BOOL "" FORCE)
add_subdirectory(deps/zstd/build/cmake EXCLUDE_FROM_ALL)
add_library(line_core STATIC src/replay.cpp)
target_include_directories(line_core PUBLIC src third_party deps/zstd/lib)
target_link_libraries(line_core PUBLIC libzstd_static)
if(MSVC)
  target_compile_options(line_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(line_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(line_tests tests/core_test.cpp)
target_link_libraries(line_tests PRIVATE line_core)
if(MSVC)
  target_compile_options(line_tests PRIVATE /UNDEBUG /utf-8)
else()
  target_compile_options(line_tests PRIVATE -UNDEBUG -Wall -Wextra)
endif()
add_test(NAME replay_and_routes COMMAND line_tests ${CMAKE_CURRENT_SOURCE_DIR}/deploy/addons/routeline/config.json)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME sr_selection_sqlite COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/sql_test.py $<TARGET_FILE:line_tests>)
add_executable(visibility_test tests/visibility_test.cpp)
target_include_directories(visibility_test PRIVATE src)
if(MSVC)
  target_compile_options(visibility_test PRIVATE /UNDEBUG)
else()
  target_compile_options(visibility_test PRIVATE -UNDEBUG -Wall -Wextra)
endif()
add_test(NAME cs2kz_particle_visibility COMMAND visibility_test)
add_executable(replay_inspect src/inspect.cpp)
target_link_libraries(replay_inspect PRIVATE line_core)

if(LINE_BUILD_PLUGIN)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The plugin targets Linux x86_64. Use WSL or Docker; core tools also support Windows.")
  endif()
  if(NOT EXISTS "${MMS}/third_party/khook/include/khook.hpp")
    message(FATAL_ERROR "Metamod API18/KHook headers missing. Run bootstrap.sh.")
  endif()
  file(READ "${SDK}/public/engine2/ivengineserver2.h" engine_header)
  if(NOT engine_header MATCHES "virtual bool[ \t]+IsSpawnGroupLoadedForClient")
    message(FATAL_ERROR "S2SDK predates the 2026-10-23 engine vtable update.")
  endif()
  file(READ "${SDK}/public/networksystem/inetworkserializer.h" netmessage_header)
  if(NOT netmessage_header MATCHES "m_pfnAllocateMessage")
    message(FATAL_ERROR "S2SDK lacks the CS2 14197 net message layout. Run bootstrap.sh to fetch the pinned zer0k-z SDK.")
  endif()
  execute_process(COMMAND ${SDK}/devtools/bin/linux/protoc
    --proto_path=${SDK}/common --proto_path=${SDK}/game/shared --proto_path=${SDK}/thirdparty/protobuf-3.21.8/src
    --cpp_out=${CMAKE_CURRENT_BINARY_DIR}
    ${SDK}/common/network_connection.proto ${SDK}/game/shared/usermessages.proto
    ${SDK}/common/networkbasetypes.proto ${SDK}/common/valveextensions.proto RESULT_VARIABLE proto_result)
  if(NOT proto_result EQUAL 0)
    message(FATAL_ERROR "SDK protoc failed: ${proto_result}")
  endif()
  add_library(routeline SHARED src/plugin.cpp
    ${CMAKE_CURRENT_BINARY_DIR}/network_connection.pb.cc ${CMAKE_CURRENT_BINARY_DIR}/usermessages.pb.cc
    ${CMAKE_CURRENT_BINARY_DIR}/networkbasetypes.pb.cc ${CMAKE_CURRENT_BINARY_DIR}/valveextensions.pb.cc
    ${SDK}/entity2/entityidentity.cpp ${SDK}/entity2/entitysystem.cpp ${SDK}/entity2/entitykeyvalues.cpp
    ${SDK}/tier0_perproject/keyvalues3.cpp ${SDK}/tier0_perproject/convar.cpp)
  target_compile_definitions(line_core PRIVATE _GLIBCXX_USE_CXX11_ABI=0)
  target_compile_definitions(routeline PRIVATE LINUX POSIX _LINUX GNUC COMPILER_GCC PLATFORM_64BITS
    _GLIBCXX_USE_CXX11_ABI=0 META_IS_SOURCE2 stricmp=strcasecmp strnicmp=strncasecmp)
  # The core and all consumers must agree on the libstdc++ string ABI.
  target_compile_definitions(line_core INTERFACE _GLIBCXX_USE_CXX11_ABI=0)
  target_compile_options(routeline PRIVATE -Wall -Wextra -Wno-unused-parameter -fvisibility=hidden -fno-gnu-unique)
  target_include_directories(routeline PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
  target_include_directories(routeline SYSTEM PRIVATE third_party/api ${MMS}/core ${MMS}/third_party/khook/include
    ${SDK}/public ${SDK}/public/tier0 ${SDK}/public/tier1 ${SDK}/public/entity2 ${SDK}/public/interfaces
    ${SDK}/game/shared ${SDK}/common ${SDK}/public/mathlib ${SDK}/thirdparty/protobuf-3.21.8/src)
  target_link_libraries(routeline PRIVATE line_core ${SDK}/lib/linux64/interfaces.a ${SDK}/lib/linux64/mathlib.a ${SDK}/lib/linux64/libtier0.so
    ${SDK}/lib/linux64/libsteam_api.so ${SDK}/lib/linux64/release/libprotobuf.a dl pthread)
  target_link_options(routeline PRIVATE -Wl,--exclude-libs,libprotobuf.a:libline_core.a:libzstd.a:mathlib.a -Wl,-z,defs -Wl,-Bsymbolic)
  set_target_properties(routeline PROPERTIES PREFIX "" BUILD_WITH_INSTALL_RPATH TRUE
    INSTALL_RPATH "$ORIGIN/../../../../../bin/linuxsteamrt64")
  add_test(NAME metamod_binary COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/binary_test.py $<TARGET_FILE:routeline>)
  set_tests_properties(metamod_binary PROPERTIES ENVIRONMENT "LD_LIBRARY_PATH=${SDK}/lib/linux64")
  add_executable(cs2kz_config_test tests/cs2kz_config_test.cpp)
  target_compile_definitions(cs2kz_config_test PRIVATE $<TARGET_PROPERTY:routeline,COMPILE_DEFINITIONS>)
  target_compile_options(cs2kz_config_test PRIVATE -UNDEBUG)
  target_include_directories(cs2kz_config_test PRIVATE src)
  target_include_directories(cs2kz_config_test SYSTEM PRIVATE $<TARGET_PROPERTY:routeline,INCLUDE_DIRECTORIES>)
  target_link_libraries(cs2kz_config_test PRIVATE ${SDK}/lib/linux64/libtier0.so)
  add_test(NAME cs2kz_original_config COMMAND cs2kz_config_test)
  set_tests_properties(cs2kz_original_config PROPERTIES ENVIRONMENT "LD_LIBRARY_PATH=${SDK}/lib/linux64")
  install(TARGETS routeline LIBRARY DESTINATION addons/routeline/bin/linuxsteamrt64)
  install(FILES deploy/addons/metamod/routeline.vdf DESTINATION addons/metamod)
  install(FILES deploy/addons/routeline/config.json deploy/addons/routeline/gamedata.json
    DESTINATION addons/routeline)
  install(FILES deploy/cfg/routeline.cfg DESTINATION cfg)
endif()
